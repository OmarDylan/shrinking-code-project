Earlier skill remember specification (documentation):
Remember

Requirements 1–2. Read fully observed train and test CSVs. Compute each goal's
training mean and sample SD (n-1 denominator), then normalize the value v:
`z = clamp((v - mean) / (sd + 1e-32), -3, 3)`;
`norm(v) = 1 / (1 + exp(-1.702 * z))`.
The heaven value h is 0 for Lbs- and 1 for Acc+, Mpg+.
`d2h = sqrt(sum((norm(v)-h)^2 for the three goals) / 3)`.
Lower is better. Use the observed complete goals for this retrospective task.
Rank by unrounded score then original index. Return all test scores in source order,
the first ten ranked indices, and the best, upper-middle (rank len(test)//2), worst.

Command: `python3 main.py remember --train TRAIN --test TEST`
Output: `{ "scores": [number,...], "top": [index,...], "best": index, "median": index, "worst": index }`.

Earlier skill certify specification (documentation):
Certify

Requirements 5–6. Compare supplied numeric distributions, each containing >=2 values.
Let n=len(xs), m=len(ys). Use sample variances vx, vy (n-1, m-1 denominators).
Pooled variance = ((n-1)*vx + (m-1)*vy)/(n+m-2).
Cohen = abs(mean(xs)-mean(ys))/(sqrt(pooled variance)+1e-32).
For constant samples the denominator convention is still 1e-32, even when the
means differ. Do not replace this with another finite sentinel or infinity.
Cliffs = abs(sum(sign(x-y) over every x in xs and y in ys))/(n*m); ties count 0.
KS = max_v |count(xs<=v)/n - count(ys<=v)/m| / sqrt(1/n+1/m), evaluated
at every distinct observed value in the union; include all ties at that value.
Same is true exactly when Cohen <=0.35 AND Cliffs <=0.197 AND KS <=1.36.
These thresholds preserve the r01 evaluator; rather than change them post hoc, any
separate cross-treatment statistical method will be explicitly labeled.

Command: `python3 main.py certify --train TRAIN --test CASES_JSON`
Input: `[{"xs":[number,...],"ys":[number,...]},...]`.
Output: `{"comparisons":[{"cohen":number,"cliffs":number,"ks":number,"same":boolean},...]}`.
One comparison per case, in supplied order. Numeric comparison tolerances are 1e-9
relative and absolute. No model training is necessary.

Earlier skill blame specification (documentation):
Blame

Use Remember's normalization for goals AND for each numeric input (that input's

import sys, csv, math, json
from bisect import bisect_left, bisect_right

GOALS = [("Lbs-", 0.0), ("Acc+", 1.0), ("Mpg+", 1.0)]


def load(path):
    with open(path, newline="") as f:
        r = csv.reader(f)
        header = [h.strip() for h in next(r)]
        rows = [[c.strip() for c in row] for row in r if row]
    return header, rows


def args():
    a = sys.argv[1:]
    d = {}
    for i, x in enumerate(a):
        if x in ("--train", "--test") and i + 1 < len(a):
            d[x[2:]] = a[i + 1]
    return a[0], d


def goal_cols(h):
    return [h.index(g) for g, _ in GOALS]


def goal_vals(rows, idx, skip_missing):
    out = []
    for row in rows:
        v = [row[i] for i in idx]
        if skip_missing and "?" in v:
            continue
        out.append([float(x) for x in v])
    return out


def goal_stats(tr):
    n = len(tr)
    stats = []
    for j in range(3):
        col = [r[j] for r in tr]
        m = sum(col) / n
        sd = math.sqrt(sum((x - m) ** 2 for x in col) / (n - 1)) if n > 1 else 0.0
        stats.append((m, sd))
    return stats


def norm(v, st):
    m, sd = st
    z = max(-3.0, min(3.0, (v - m) / (sd + 1e-32)))
    return 1 / (1 + math.exp(-1.702 * z))


def d2h(r, stats):
    s = 0.0
    for j in range(3):
        nv = norm(r[j], stats[j])
        s += (nv - GOALS[j][1]) ** 2
    return math.sqrt(s / 3)


def remember(d):
    h, rows = load(d["train"])
    stats = goal_stats(goal_vals(rows, goal_cols(h), True))
    h2, rows2 = load(d["test"])
    te = goal_vals(rows2, goal_cols(h2), False)
    scores = [d2h(r, stats) for r in te]
    order = sorted(range(len(scores)), key=lambda i: (scores[i], i))
    print(json.dumps({
        "scores": scores,
        "top": order[:10],
        "best": order[0],
        "median": order[len(order) // 2],
        "worst": order[-1],
    }))


def num(x):
    try:
        v = float(x)
        return v if math.isfinite(v) else float("nan")
    except ValueError:
        return float("nan")


def predict(d):
    import numpy as np
    from sklearn.ensemble import RandomForestRegressor
    from sklearn.impute import SimpleImputer
    from sklearn.pipeline import make_pipeline

    h, rows = load(d["train"])
    h2, rows2 = load(d["test"])
    gi = goal_cols(h)
    stats = goal_stats(goal_vals(rows, gi, True))
    cats = sorted({r[h.index("origin")] for r in rows} - {"?"})

    def feats(hh, rs):
        ci, vi, mi, oi = (hh.index(c) for c in ("Clndrs", "Volume", "Model", "origin"))
        return np.array([[num(r[ci]), num(r[vi]), num(r[mi])] +
                         [1.0 if r[oi] == c else 0.0 for c in cats] for r in rs])

    Xtr, Xte = feats(h, rows), feats(h2, rows2)
    mpg = np.array([num(r[gi[2]]) for r in rows])
    dh = np.array([d2h([num(r[i]) for i in gi], stats)
                   if all(math.isfinite(num(r[i])) for i in gi) else float("nan")
                   for r in rows])

    def fit_pred(y, cols, rowsel):
        ok = np.isfinite(y)
        if ok.sum() < 2:
            return np.full(int(rowsel.sum()), float(np.nanmean(y)) if ok.any() else 0.5)
        m = make_pipeline(SimpleImputer(strategy="median"),
                          RandomForestRegressor(300, min_samples_leaf=2, random_state=0))
        m.fit(Xtr[ok][:, cols], y[ok])
        return m.predict(Xte[rowsel][:, cols])

    n = len(rows2)
    allc = list(range(Xtr.shape[1]))
    noVol = [c for c in allc if c != 1]
    miss = np.array([not math.isfinite(Xte[i, 1]) for i in range(n)], dtype=bool)
    out_m = np.zeros(n)
    out_d = np.zeros(n)
    for sel, cols in ((~miss, allc), (miss, noVol)):
        if sel.any():
            out_m[sel] = fit_pred(mpg, cols, sel)
            out_d[sel] = fit_pred(dh, cols, sel)
    mp = [max(float(x), 1e-3) for x in out_m]
    sc = [min(1.0, max(0.0, float(x))) for x in out_d]
    return mp, sc


def guess(d):
    mp, sc = predict(d)
    print(json.dumps({"mpg": mp, "scores": sc}))


def compare(xs, ys):
    xs = [float(x) for x in xs]
    ys = [float(y) for y in ys]
    n, m = len(xs), len(ys)
    mx, my = math.fsum(xs) / n, math.fsum(ys) / m
    vx = math.fsum((x - mx) ** 2 for x in xs) / (n - 1)
    vy = math.fsum((y - my) ** 2 for y in ys) / (m - 1)
    pooled = ((n - 1) * vx + (m - 1) * vy) / (n + m - 2)
    cohen = abs(mx - my) / (math.sqrt(pooled) + 1e-32)
    sx, sy = sorted(xs), sorted(ys)
    gt = sum(bisect_left(sy, x) for x in xs)
    lt = sum(m - bisect_right(sy, x) for x in xs)
    cliffs = abs(gt - lt) / (n * m)
    ks = 0.0
    for v in set(xs) | set(ys):
        ks = max(ks, abs(bisect_right(sx, v) / n - bisect_right(sy, v) / m))
    ks /= math.sqrt(1 / n + 1 / m)
    same = cohen <= 0.35 and cliffs <= 0.197 and ks <= 1.36
    return {"cohen": cohen, "cliffs": cliffs, "ks": ks, "same": bool(same)}


def certify(d):
    with open(d["test"]) as f:
        cases = json.load(f)
    print(json.dumps({"comparisons": [compare(c["xs"], c["ys"]) for c in cases]}))


def flow(d):
    h, rows = load(d["train"])
    gv = goal_vals(rows, goal_cols(h), True)
    stats = goal_stats(gv)
    hist = [d2h(r, stats) for r in gv]
    mp, sc = predict(d)
    alerts = []
    for i in range(len(sc)):
        if i < 19:
            alerts.append(False)
        else:
            alerts.append(not compare(hist, sc[i - 19:i + 1])["same"])
    print(json.dumps({"mpg": mp, "scores": sc, "alerts": alerts}))


def blame(d):
    h, rows = load(d["train"])
    gi = goal_cols(h)
    rows = [r for r in rows if all(r[i] != "?" for i in gi)]
    gv = [[float(r[i]) for i in gi] for r in rows]
    stats = goal_stats(gv)
    sc = [d2h(g, stats) for g in gv]
    order = sorted(range(len(rows)), key=lambda i: (round(sc[i], 12), i))
    q = max(1, len(rows) // 4)
    good = [rows[i] for i in order[:q]]
    poor = [rows[i] for i in order[-q:]]

    nums = ["Clndrs", "Volume", "Model"]
    nstat, centers = {}, {}
    for f in nums:
        c = h.index(f)
        col = [float(r[c]) for r in rows if r[c] != "?"]
        n = len(col)
        m = sum(col) / n if n else 0.0
        sd = math.sqrt(sum((x - m) ** 2 for x in col) / (n - 1)) if n > 1 else 0.0
        nstat[f] = (m, sd)
        cs = []
        for grp in (good, poor):
            vals = [float(r[c]) for r in grp if r[c] != "?"]
            cs.append(norm(sum(vals) / len(vals), nstat[f]) if vals else None)
        centers[f] = cs
    oc = h.index("origin")

    def freq(grp):
        vals = [r[oc] for r in grp if r[oc] != "?"]
        out = {}
        for v in vals:
            out[v] = out.get(v, 0) + 1 / len(vals)
        return out
    gf, pf = freq(good), freq(poor)

    h2, rows2 = load(d["test"])
    g2 = goal_cols(h2)
    res = []
    for r in rows2:
        feats = {}
        for f in nums:
            x = r[h2.index(f)]
            gc, pc = centers[f]
            if x == "?" or gc is None or pc is None:
                feats[f] = 0.0
            else:
                nx = norm(float(x), nstat[f])
                feats[f] = max(0.0, abs(nx - gc) - abs(nx - pc))
        x = r[h2.index("origin")]
        feats["origin"] = 0.0 if x == "?" else max(0.0, pf.get(x, 0.0) - gf.get(x, 0.0))
        goals = {}
        for j, (g, hv) in enumerate(GOALS):
            x = r[g2[j]]
            goals[g] = 0.0 if x == "?" else abs(norm(float(x), stats[j]) - hv)
        key = lambda s: (lambda n: (-round(s[n], 10), n))
        res.append({
            "goals": goals,
            "features": feats,
            "goal_order": sorted(goals, key=key(goals)),
            "feature_order": sorted(feats, key=key(feats)),
        })
    print(json.dumps({"diagnoses": res}))


def main():
    if not sys.argv[1:]:
        sys.exit(1)
    cmd, d = args()
    if cmd == "remember":
        remember(d)
    elif cmd == "guess":
        guess(d)
    elif cmd == "certify":
        certify(d)
    elif cmd == "flow":
        flow(d)
    elif cmd == "blame":
        blame(d)
    else:
        sys.exit(1)


main()
